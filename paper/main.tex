\section{Introduction}
\IEEEPARstart{T}{his} opening paragraph provides the general background of
the work: the application scenario, the driving trend, and why the problem
is worth studying now. Keep it accessible to the general reader of the
proceedings and quantify the impact with one or two illustrative figures
of merit, then narrow down to the specific problem addressed in this
paper.

The second paragraph reviews the state of the art at a high level and
identifies the gap. Summarize the dominant existing approaches in one or
two sentences each, explain their common limitation, and state explicitly
what is still missing. This motivates the contributions of this paper,
typically summarized as a bullet list:
\begin{itemize}
\item We formulate ... as ...
\item We propose a ... algorithm to ...
\item Theoretical analysis and simulation results demonstrate that ...
\end{itemize}

The closing paragraph of the introduction states the headline result in
one or two sentences, e.g., ``the proposed scheme achieves a 20\% gain
over the strongest baseline under typical operating conditions,'' which
anchors the contribution list above.

\section{Problem Formulation}

\begin{subequations}\label{opt:main}
\begin{align}
\min_{\mathbf{x}} \quad & f_0(\mathbf{x}) \label{opt:obj}\\
\text{s.t.}\quad
& f_i(\mathbf{x}) \le 0, \quad i = 1, \ldots, m, \label{opt:ineq}\\
& h_j(\mathbf{x}) = 0, \quad j = 1, \ldots, p. \label{opt:eq}
\end{align}
\end{subequations}
Explain each line of problem~\eqref{opt:main} in words: the meaning of the

\subsection{Iterative Update}
Detail one iteration of Algorithm~\ref{alg:proposed}. With the other
blocks fixed, the update of block $k$ reduces to a subproblem of the form
\begin{equation}\label{eq:update}
\mathbf{x}_{k}^{(t+1)} \in \arg\min_{\mathbf{x}_{k}}\;
f_0\big(\mathbf{x}_{k}, \mathbf{x}_{-k}^{(t)}\big),
\end{equation}
which admits a closed-form or low-complexity solution because ...
Repeat the argument for the remaining blocks and state the cost of one
full iteration.

\begin{algorithm}[!t]
\caption{Proposed Algorithm}
\label{alg:proposed}
\begin{algorithmic}[1]
\STATE Initialize $\mathbf{x}^{(0)}$ and set $t = 0$
\REPEAT
\STATE Update one block of $\mathbf{x}$ with the other blocks fixed
\STATE $t \leftarrow t + 1$
\UNTIL convergence in objective~\eqref{opt:obj}
\STATE \textbf{return} $\mathbf{x}^{(t)}$
\end{algorithmic}
\end{algorithm}

\subsection{Convergence and Complexity}

stationary point under suitable conditions, and bound the overall
complexity in terms of the problem dimensions and the number of
iterations. Formal proofs are deferred to an appendix or omitted for
brevity.
